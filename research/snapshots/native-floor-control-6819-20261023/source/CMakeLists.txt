cmake_minimum_required(VERSION 3.20)
project(CitadelCastNativeWallQuery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../unlocker-mount-6817-20261022/sdk")
function(sdk_target target)
 target_compile_definitions(${target} PRIVATE COMPILER_MSVC COMPILER_MSVC64 PLATFORM_64BITS X64BITS WIN32 WINDOWS CRT_SECURE_NO_WARNINGS CRT_SECURE_NO_DEPRECATE CRT_NONSTDC_NO_DEPRECATE _MBCS)
 target_include_directories(${target} PRIVATE "${SDK}" "${SDK}/thirdparty/protobuf-3.21.8/src" "${SDK}/common" "${SDK}/game/shared" "${SDK}/game/server" "${SDK}/public" "${SDK}/public/engine" "${SDK}/public/mathlib" "${SDK}/public/tier0" "${SDK}/public/tier1" "${SDK}/public/entity2")
endfunction()
add_library(server SHARED main.cpp cvars.cpp interfaces.cpp adapter.cpp "${SDK}/tier1/convar.cpp")
sdk_target(server)
target_compile_definitions(server PRIVATE CreateInterface=_CreateInterface)
target_link_libraries(server PRIVATE "${SDK}/lib/public/win64/tier0.lib" "${SDK}/lib/public/win64/interfaces.lib" "${SDK}/lib/public/win64/mathlib.lib" "${SDK}/lib/public/win64/2015/libprotobuf.lib" bcrypt.lib)
target_compile_options(server PRIVATE /fp:precise)
enable_testing()
foreach(name IN ITEMS wall_native query_contract lifecycle owned_slot hero_target page_read)
 add_executable(${name}_tests test_${name}.cpp)
 target_compile_options(${name}_tests PRIVATE /UNDEBUG /fp:precise)
 if(name STREQUAL "wall_native")
  sdk_target(${name}_tests)
 endif()
 if(name STREQUAL "page_read")
  add_test(NAME ${name} COMMAND ${name}_tests "${CMAKE_CURRENT_BINARY_DIR}/page-read-test.json")
 else()
  add_test(NAME ${name} COMMAND ${name}_tests)
 endif()
endforeach()
add_executable(live_nohit_tests test_live_nohit.cpp)
target_compile_options(live_nohit_tests PRIVATE /fp:precise)
add_test(NAME saved_native_nohit_fixture COMMAND live_nohit_tests)
add_executable(floor_control_tests test_floor_control.cpp)
target_compile_options(floor_control_tests PRIVATE /fp:precise)
add_test(NAME floor_control COMMAND floor_control_tests)
