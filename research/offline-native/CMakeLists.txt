cmake_minimum_required(VERSION 3.20)
project(CitadelCastOfflineNativeChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "These owned-allocation Windows fixtures require MSVC and the Windows SDK.")
endif()
enable_testing()
set(CAMERA "${CMAKE_CURRENT_LIST_DIR}/../snapshots/hero-framing-ab-6819-20261023")
set(CONTACT "${CMAKE_CURRENT_LIST_DIR}/../snapshots/native-floor-control-6819-20261023")

function(camera_check name)
  add_executable(${name} "${CAMERA}/source/test_${name}.cpp")
  target_compile_options(${name} PRIVATE /UNDEBUG /fp:precise)
  add_test(NAME camera_${name} COMMAND ${name} ${ARGN})
endfunction()
